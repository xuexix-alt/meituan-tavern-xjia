# ZOD MVU 店铺列表实战案例

## 📋 需求分析

用户希望创建一个全局变量`店铺列表`，包含多个结构一致的店铺信息，每个店铺包含多个套餐，每个套餐包含详细的描述和图片信息。

---

## 🎯 最佳实践方案

### 1. Schema设计（变量结构.js）

```javascript
import { registerMvuSchema } from 'https://testingcf.jsdelivr.net/gh/StageDog/tavern_resource/dist/util/mvu_zod.js';

export const Schema = z.object({
  // ===== 全局变量：店铺列表 =====
  // 使用 z.record() 通过 shop_id 直接访问店铺
  店铺列表: z.record(
    z.object({
      // 店铺基本信息
      shopname: z.string(),  // 店铺名称
      shop_id: z.coerce.number().int().positive(),  // 店铺ID（作为键）
      shoptags: z.array(z.string()),  // 店铺标签数组

      // 套餐列表（使用 record 以套餐名称为键）
      packages: z.record(
        z.object({
          // 套餐基本信息
          name: z.string(),  // 套餐名称
          price: z.coerce.number().int().positive(),  // 价格（转换为数字）
          stars: z.coerce.number().transform(v => _.clamp(v, 0, 5)),  // 评分（0-5）
          icon: z.string(),  // FontAwesome图标类名

          // 标签
          tags: z.array(z.string()),  // 套餐标签数组

          // 图片描述（AI生成的图片prompt）
          image1: z.string(),
          image2: z.string(),
          image3: z.string(),

          // 文本描述
          description: z.string(),
          content: z.array(z.string()),  // 套餐内容描述数组
          reviews: z.array(z.string())  // 评价数组

        }).strict()
      ).prefault({})  // 默认空对象，避免未定义错误

    }).strict()
  ).prefault({})  // 默认空对象

}).strict();

$(() => {
  registerMvuSchema(Schema);
})
```

### 3. 更新规则（变量更新规则.yaml）

```yaml
---
变量更新规则:
  # ===== 店铺列表规则 =====
  店铺列表:
    type: |-
      {
        [shop_id: string]: {  // 以shop_id为键
          shopname: string;
          shop_id: number;
          shoptags: string[];
          packages: {
            [package_name: string]: {  // 以套餐名称为键
              name: string;
              price: number;
              stars: number;
              icon: string;
              tags: string[];
              image1: string;
              image2: string;
              image3: string;
              description: string;
              content: string[];
              reviews: string[];
            }
          }
        }
      }
    check:
      - 任何角色或系统都可以查看店铺列表
      - 只有系统管理员可以添加、修改或删除店铺
      - 店铺ID（shop_id）必须唯一且为正整数
      - 套餐名称在同一店铺内必须唯一

  # 店铺级别操作
  "店铺列表/{shop_id}":
    shopname:
      type: string
      check:
        - 添加新店铺时设置
        - 店铺重命名时更新
        - 名称应简洁明确，突出特色

    shop_id:
      type: number
      check:
        - 作为店铺的唯一标识符
        - 必须是正整数且唯一
        - 创建后不建议修改

    shoptags:
      type: string[]
      check:
        - 标签数量建议3-5个
        - 标签应准确描述店铺特色
        - 可根据店铺定位调整

  # 套餐级别操作
  "店铺列表/{shop_id}/packages/{package_name}":
    price:
      type: number
      range: 1~999999
      check:
        - 设置套餐价格（单位：虚拟货币）
        - 价格应与套餐内容匹配
        - 特殊活动可以临时调整

    stars:
      type: number
      range: 0~5
      check:
        - 用户评价后的平均分
        - 支持小数（如4.8分）
        - 自动计算或手动更新

    tags:
      type: string[]
      check:
        - 套餐标签描述核心特色
        - 标签数量建议2-4个
        - 用于筛选和推荐

    description:
      type: string
      check:
        - 详细描述套餐内容和特色
        - 字数建议200-500字
        - 应吸引用户购买

    content:
      type: string[]
      check:
        - 列举套餐包含的具体内容
        - 每个内容项应简洁明确
        - 建议3-5项

    reviews:
      type: string[]
      check:
        - 用户评价和反馈
        - 可以包含正面和负面评价
        - 用于展示真实性和参考
```

    const all_variables = getAllVariables();
    const characterData = all_variables.stat_data;
    const shopList = _.get(characterData, '店铺列表', {});

    // 清空现有内容
    $('#shop-list-container').empty();

    // 遍历店铺
    _.forEach(shopList, function(shop, shopId) {

                    <h3 class="shop-name">${shop.shopname}</h3>

                </div>
                <div class="shop-tags">
                    ${shop.shoptags.map(tag => `<span class="tag">${tag}</span>`).join('')}
                </div>
                <div class="packages-section">

## 💡 设计优势

### 1. **使用 z.record() 的优势**

```javascript
// ✅ 推荐：通过键直接访问
店铺列表: z.record(
  z.object({...})  // shop_id作为键
)

// ❌ 避免：使用数组（难以定位）
店铺列表: z.array(
  z.object({
    shop_id: z.number(),
    ...
  })
)
```

**优势**：
- 通过 `shop_id` 直接访问特定店铺：`店铺列表/88231923`
- 通过套餐名称直接访问：`店铺列表/88231923/packages/套餐名`
- 避免遍历数组查找的复杂性

### 2. **类型安全保证**

```javascript
// 价格自动转换为数字
price: z.coerce.number().int().positive(),

// 评分自动限制范围
stars: z.coerce.number().transform(v => _.clamp(v, 0, 5)),

// 自动类型转换
shop_id: z.coerce.number().int().positive(),
```

### 3. **幂等性保证**

```javascript
// 多次添加同一套餐不会重复
packages: z.record(z.object({...})).prefault({})

// 重复解析保持一致
Schema.parse(Schema.parse(data)) === Schema.parse(data)
```
